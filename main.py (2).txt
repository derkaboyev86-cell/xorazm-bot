def ai_api_answer_sync(question):
    if not GROQ_API_KEY:
        return "❌ GROQ_API_KEY yo'q! Render.com da Environment Variables ga qo'shing"
    try:
        url = "https://api.groq.com/openai/v1/chat/completions"
        headers = {"Authorization": f"Bearer {GROQ_API_KEY}", "Content-Type": "application/json"}
        models = ["openai/gpt-oss-20b","openai/gpt-oss-120b","llama-3.3-70b-versatile","gemma2-9b-it"]
        for model_name in models:
            try:
                payload = {
                    "model": model_name,
                    "messages": [
                        {"role": "system", "content": "Sen islomiy savollarga Qur'on va sunnat asosida, boshqa savollarga ham foydali javob beradigan, o'zbek lotin alifbosida yozadigan yordamchisan."},
                        {"role": "user", "content": question}
                    ],
                    "temperature": 0.8,
                    "max_tokens": 1000
                }
                r = requests.post(url, json=payload, headers=headers, timeout=30)
                j = r.json()
                if "choices" in j and j["choices"]:
                    return j["choices"][0]["message"]["content"]
            except:
                continue
        return "⚠️ AI javob bera olmadi."
    except Exception as e:
        return f"⚠️ Xato: {e}"

async def location_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.message.location:
        lat=update.message.location.latitude; lon=update.message.location.longitude
        ka_lat=21.4225; ka_lon=39.8262
        dLon=math.radians(ka_lon-lon); lat1=math.radians(lat); lat2=math.radians(ka_lat)
        y=math.sin(dLon)*math.cos(lat2); x=math.cos(lat1)*math.sin(lat2)-math.sin(lat1)*math.cos(lat2)*math.cos(dLon)
        qib=(math.degrees(math.atan2(y,x))+360)%360
        await update.message.reply_text(f"Qibla: {qib:.1f}°")

async def text_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.message.location:
        return
    txt = update.message.text
    if not txt or txt.startswith('/'):
        return
    try:
        await context.bot.send_chat_action(chat_id=update.effective_chat.id, action="typing")
    except:
        pass
    answer = ai_api_answer_sync(txt)
    try:
        await update.message.reply_text(answer)
    except Exception as e:
        await update.message.reply_text(f"Xato: {e}")

    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, text_handler))
    print("✅ BOT TO'LIQ - SEN BERGAN MODEL openai/gpt-oss-20b - 24/7 READY")
    app.run_polling()

if __name__=="__main__":
